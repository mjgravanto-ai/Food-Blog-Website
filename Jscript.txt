const postList = document.getElementById("postList");
const searchInput = document.getElementById("searchInput");
const resultsNote = document.getElementById("resultsNote");
const emptyState = document.getElementById("emptyState");
const dialog = document.getElementById("recipeDialog");
const dialogContent = document.getElementById("dialogContent");
let activeCategory = "All";
let searchTerm = "";

function renderRecipes() {
  const filtered = recipes.filter((recipe) => {
    const matchesCategory = activeCategory === "All" || recipe.category === activeCategory;
    const haystack = `${recipe.title} ${recipe.category} ${recipe.description} ${recipe.ingredients.join(" ")}`.toLowerCase();
    return matchesCategory && haystack.includes(searchTerm.trim().toLowerCase());
  });

  postList.innerHTML = filtered.map((recipe, index) => `
    <article class="post-row" style="animation-delay:${Math.min(index * 45, 180)}ms">
      <button class="post-image-button" type="button" data-recipe="${recipe.id}" aria-label="Read ${recipe.title}"><img src="${recipe.image}" alt="${recipe.alt}" loading="lazy" /></button>
      <div class="post-copy">
        <span class="post-category">${recipe.category}</span>
        <button class="post-title" type="button" data-recipe="${recipe.id}">${recipe.title}</button>
        <p class="post-description">${recipe.description}</p>
        <div class="post-meta"><span>${recipe.time}</span><span class="meta-dot" aria-hidden="true">·</span><span>Made for sharing</span></div>
      </div>
      <button class="read-more" type="button" data-recipe="${recipe.id}">Read the recipe <span aria-hidden="true">→</span></button>
    </article>`).join("");

  emptyState.hidden = filtered.length !== 0;
  const isFiltered = activeCategory !== "All" || searchTerm.trim() !== "";
  resultsNote.textContent = isFiltered ? `${filtered.length} ${filtered.length === 1 ? "recipe" : "recipes"}${searchTerm.trim() ? ` matching “${searchTerm.trim()}”` : ""}${activeCategory !== "All" ? ` in ${activeCategory.toLowerCase()}` : ""}` : "";
}

function openRecipe(id) {
  const recipe = recipes.find((item) => item.id === id);
  if (!recipe) return;
  dialogContent.innerHTML = `
    <img class="dialog-image" src="${recipe.image}" alt="${recipe.alt}" />
    <div class="dialog-body">
      <span class="post-category">${recipe.category} <span aria-hidden="true">·</span> ${recipe.time}</span>
      <h2 id="dialogTitle">${recipe.title}</h2>
      <p>${recipe.description}</p>
      <div class="dialog-columns">
        <section><h3>Gather your ingredients</h3><ul>${recipe.ingredients.map((item) => `<li>${item}</li>`).join("")}</ul></section>
        <section><h3>Make it with love</h3><ol>${recipe.instructions.map((item) => `<li>${item}</li>`).join("")}</ol></section>
      </div>
    </div>`;
  dialog.showModal();
}

postList.addEventListener("click", (event) => {
  const trigger = event.target.closest("[data-recipe]");
  if (trigger) openRecipe(trigger.dataset.recipe);
});

document.getElementById("featuredReadMore").addEventListener("click", () => openRecipe("pork-adobo"));
document.getElementById("dialogClose").addEventListener("click", () => dialog.close());
dialog.addEventListener("click", (event) => { if (event.target === dialog) dialog.close(); });
searchInput.addEventListener("input", () => { searchTerm = searchInput.value; renderRecipes(); });
document.getElementById("searchForm").addEventListener("submit", (event) => event.preventDefault());
document.getElementById("clearSearch").addEventListener("click", () => {
  searchInput.value = "";
  searchTerm = "";
  activeCategory = "All";
  document.querySelectorAll(".category-link").forEach((button) => button.classList.toggle("active", button.dataset.category === "All"));
  renderRecipes();
});

document.getElementById("categoryNav").addEventListener("click", (event) => {
  const button = event.target.closest("[data-category]");
  if (!button) return;
  activeCategory = button.dataset.category;
  document.querySelectorAll(".category-link").forEach((item) => item.classList.toggle("active", item === button));
  renderRecipes();
  document.getElementById("recipes").scrollIntoView({ behavior: "smooth", block: "start" });
});

const menuToggle = document.getElementById("menuToggle");
const sidebar = document.getElementById("sidebar");
function setMenuOpen(isOpen) {
  document.body.classList.toggle("menu-open", isOpen);
  menuToggle.setAttribute("aria-expanded", String(isOpen));
  menuToggle.setAttribute("aria-label", isOpen ? "Close menu" : "Open menu");
  sidebar.setAttribute("aria-hidden", String(!isOpen));
  if (isOpen) document.getElementById("closeSidebar").focus();
  else menuToggle.focus();
}
menuToggle.addEventListener("click", () => setMenuOpen(true));
document.getElementById("closeSidebar").addEventListener("click", () => setMenuOpen(false));
document.getElementById("sidebarBackdrop").addEventListener("click", () => setMenuOpen(false));
document.addEventListener("keydown", (event) => {
  if (event.key === "Escape" && document.body.classList.contains("menu-open")) setMenuOpen(false);
  if (event.key === "/" && !["INPUT", "TEXTAREA"].includes(document.activeElement.tagName) && !dialog.open) {
    event.preventDefault();
    searchInput.focus();
  }
});
document.querySelectorAll(".side-link").forEach((link) => link.addEventListener("click", () => setMenuOpen(false)));

renderRecipes();
